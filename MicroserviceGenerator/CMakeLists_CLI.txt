cmake_minimum_required(VERSION 3.16)
project(MicroserviceGenerator)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Single executable, minimal dependencies (standard library only)
add_executable(microservice-gen
    src/main.cpp
)

# Compiler optimizations
if(MSVC)
    target_compile_options(microservice-gen PRIVATE /O2)
else()
    target_compile_options(microservice-gen PRIVATE -O2)
endif()

# Enable static linking for standalone executable
if(MSVC)
    set_property(TARGET microservice-gen PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_options(microservice-gen PRIVATE /SUBSYSTEM:CONSOLE)
endif()

# Output settings
set_target_properties(microservice-gen PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)

# For cross-platform binary name
if(WIN32)
    set_target_properties(microservice-gen PROPERTIES
        SUFFIX ".exe"
    )
endif()
